## Naming decision (lock this in before Task 1)

The master already uses British `colour_type` and `leafing_colour` alongside legacy American `colors`.
New columns use **British** spelling. `products.colors` keeps its meaning: **total printing colours**.

| Column | Type | Meaning |
|---|---|---|
| `colour_type` *(exists)* | TEXT | `CMYK` \| `Pantone` \| `CMYK + Pantone` |
| `colors` *(exists)* | INTEGER | Total printing colours (CMYK + Pantone + metallic) |
| `print_process` | TEXT | `Offset` \| `Metallic` \| `Offset + Metallic` |
| `cmyk_colours` | INTEGER | CMYK count, normally 4 |
| `pantone_colours` | INTEGER | Pantone count |
| `pantone_codes` | TEXT | Free text, e.g. `Pantone 186 C, Pantone 286 C` |
| `metallic_colours` | INTEGER | Metallic ink count |
| `metallic_details` | TEXT | e.g. `Metallic Gold (Pantone 871 C)` |
| `print_instructions` | TEXT | Special printing instructions |

**No backfill.** 300+ live products would have to be guessed at. Instead the resolvers in Task 2 infer
counts from `colour_type` + `colors` when the new columns are NULL, so every existing product reads
correctly on day one and gets exact numbers only when someone types them.

```jsx
// Printing colour + process — ONE vocabulary for the whole ERP.
//
// This is the BoardStatus.jsx contract applied to ink: the data decides the
// facts, this file decides how they LOOK, so Planning, Artwork, the Job Card,
// Print Planning and the press queue cannot describe one job five ways. A page
// that re-declares its own colour labels or tones is the bug this module exists
// to prevent.
//
// TWO AXES, never merged:
//   colour_type  — what the colour build IS   (CMYK / Pantone / CMYK + Pantone)
//   print_process— how it is LAID DOWN        (Offset / Metallic / Offset + Metallic)
// A Pantone spot colour is not a metallic ink. Folding them into one field is
// what put metallic and Pantone jobs in the same bucket on the floor.
import { Droplet, Palette, Layers, Printer, Sparkles } from 'lucide-react';

export const COLOUR_TYPES = ['CMYK', 'Pantone', 'CMYK + Pantone'];
export const PRINT_PROCESSES = ['Offset', 'Metallic', 'Offset + Metallic'];

// Short heads for filter chips (they carry a count and live in tight rails);
// the whole phrase on a badge.
export const COLOUR_LABEL = {
  'CMYK': 'CMYK', 'Pantone': 'Pantone', 'CMYK + Pantone': 'CMYK + Pantone',
};
export const PROCESS_LABEL = {
  'Offset': 'Offset', 'Metallic': 'Metallic', 'Offset + Metallic': 'Offset + Metallic',
};
export const COLOUR_HINT = {
  'CMYK': 'four-colour process only — no spot ink',
  'Pantone': 'spot colours only — no process build',
  'CMYK + Pantone': 'process build plus one or more spot colours',
};
export const PROCESS_HINT = {
  'Offset': 'conventional offset — no metallic ink or process',
  'Metallic': 'metallic ink or metallic process only',
  'Offset + Metallic': 'offset plus a metallic ink or process',
};
// Distinct but professional, and never colour ALONE — every badge carries its
// text label, because a press operator reading a printed sheet has no hue.
// Composition runs cool (sky → violet), process runs warm (slate → amber), so
// the two axes never read as one scale.
export const COLOUR_TONE = {
  'CMYK': 'border-sky-200 bg-sky-50 text-sky-700',
  'Pantone': 'border-violet-200 bg-violet-50 text-violet-700',
  'CMYK + Pantone': 'border-indigo-200 bg-indigo-50 text-indigo-700',
};
export const PROCESS_TONE = {
  'Offset': 'border-slate-200 bg-slate-50 text-slate-600',
  'Metallic': 'border-amber-300 bg-amber-100 text-amber-800',
  'Offset + Metallic': 'border-orange-300 bg-orange-100 text-orange-800',
};
export const COLOUR_COUNT_TONE = {
  'CMYK': 'bg-white/70', 'Pantone': 'bg-white/70', 'CMYK + Pantone': 'bg-white/70',
};
export const PROCESS_COUNT_TONE = {
  'Offset': 'bg-white/70', 'Metallic': 'bg-white/80', 'Offset + Metallic': 'bg-white/80',
};
export const COLOUR_ICON = { 'CMYK': Droplet, 'Pantone': Palette, 'CMYK + Pantone': Layers };
export const PROCESS_ICON = { 'Offset': Printer, 'Metallic': Sparkles, 'Offset + Metallic': Sparkles };
// Sort order: richest build first, most special process first.
export const COLOUR_RANK = { 'CMYK + Pantone': 0, 'Pantone': 1, 'CMYK': 2 };
export const PROCESS_RANK = { 'Offset + Metallic': 0, 'Metallic': 1, 'Offset': 2 };

// ---- Resolvers -------------------------------------------------------------
// Every one of these tolerates a row from an older API response mid-deploy, and
// a product nobody has filled the new fields in on. That is the normal case on
// day one: 300+ live products carry only colour_type + colors.

const norm = v => String(v ?? '').trim();

// Tolerates the free-text history in products.colour_type ("cmyk+pantone",
// "CMYK & Pantone") — the master is a picker now, but imported rows are not.
export function colourTypeOf(row) {
  const raw = norm(row?.colour_type);
  if (!raw) return null;
  const hit = COLOUR_TYPES.find(t => t.toLowerCase() === raw.toLowerCase());
  if (hit) return hit;
  const low = raw.toLowerCase();
  const hasC = low.includes('cmyk') || low.includes('process');
  const hasP = low.includes('pantone') || low.includes('spot');
  if (hasC && hasP) return 'CMYK + Pantone';
  if (hasP) return 'Pantone';
  if (hasC) return 'CMYK';
  return null;
}

export function processOf(row) {
  const raw = norm(row?.print_process);
  if (raw) {
    const hit = PRINT_PROCESSES.find(t => t.toLowerCase() === raw.toLowerCase());
    if (hit) return hit;
    const low = raw.toLowerCase();
    const hasM = low.includes('metallic');
    const hasO = low.includes('offset');
    if (hasM && hasO) return 'Offset + Metallic';
    if (hasM) return 'Metallic';
    if (hasO) return 'Offset';
  }
  // No process recorded: a metallic count or metallic detail is the only thing
  // that may imply metallic. A Pantone code NEVER does.
  const m = Number(row?.metallic_colours) || 0;
  if (m > 0 || norm(row?.metallic_details)) return 'Offset + Metallic';
  return raw ? null : 'Offset';
}

const num = v => (v == null || v === '' ? null : Number(v));

export function totalColoursOf(row) {
  const t = num(row?.colors);
  if (t != null && Number.isFinite(t)) return t;
  const parts = [row?.cmyk_colours, row?.pantone_colours, row?.metallic_colours].map(v => Number(v) || 0);
  const sum = parts.reduce((a, b) => a + b, 0);
  return sum || null;
}

export function cmykCountOf(row) {
  const explicit = num(row?.cmyk_colours);
  if (explicit != null && Number.isFinite(explicit)) return explicit;
  const type = colourTypeOf(row);
  if (!type || type === 'Pantone') return 0;
  return 4;                                  // the plant's process set is always 4
}

export function pantoneCountOf(row) {
  const explicit = num(row?.pantone_colours);
  if (explicit != null && Number.isFinite(explicit)) return explicit;
  const type = colourTypeOf(row);
  if (!type || type === 'CMYK') return 0;
  const total = totalColoursOf(row);
  if (total == null) return 0;
  const metallic = Number(row?.metallic_colours) || 0;
  if (type === 'Pantone') return Math.max(0, total - metallic);
  return Math.max(0, total - 4 - metallic);  // CMYK + Pantone
}

export function metallicCountOf(row) {
  const explicit = num(row?.metallic_colours);
  if (explicit != null && Number.isFinite(explicit)) return explicit;
  const p = processOf(row);
  if (p === 'Metallic' || p === 'Offset + Metallic') return norm(row?.metallic_details) ? 1 : 0;
  return 0;
}

// The compact sentence a table cell or export column carries:
//   "CMYK — 4 colours" · "CMYK + 2 Pantone — 6 colours" · "CMYK + Metallic Gold — 5 colours"
export function colourSummary(row) {
  const type = colourTypeOf(row);
  const total = totalColoursOf(row);
  if (!type && total == null) return '—';
  const pantone = pantoneCountOf(row);
  const metallic = metallicCountOf(row);
  const bits = [];
  if (type === 'CMYK' || type === 'CMYK + Pantone') bits.push('CMYK');
  if (pantone > 0) bits.push(`${pantone} Pantone`);
  else if (type === 'Pantone') bits.push('Pantone');
  if (metallic > 0) {
    const name = metallicNameOf(row);
    bits.push(name ? `Metallic ${name}` : `${metallic} Metallic`);
  }
  const head = bits.length ? bits.join(' + ') : (type || '—');
  return total != null ? `${head} — ${total} ${total === 1 ? 'colour' : 'colours'}` : head;
}

// First metallic shade named in metallic_details ("Metallic Gold (Pantone 871 C)" → "Gold").
export function metallicNameOf(row) {
  const d = norm(row?.metallic_details);
  if (!d) return null;
  const first = d.split(/[,;/]/)[0].trim();
  return first.replace(/^metallic\s+/i, '').replace(/\s*\(.*\)\s*$/, '').trim() || null;
}

// Everything a row should match on when someone types into a search box.
export function colourSearchText(row) {
  return [
    colourTypeOf(row), processOf(row), colourSummary(row),
    row?.pantone_codes, row?.metallic_details, row?.print_instructions,
    totalColoursOf(row) != null ? `${totalColoursOf(row)} colours` : null,
  ].filter(Boolean).join(' ');
}

// Total-colour banding, for the "1 / 2 / 3 / 4 / 5–6 / 6+" filter.
export const COLOUR_BANDS = [
  { key: '1', label: '1', test: n => n === 1 },
  { key: '2', label: '2', test: n => n === 2 },
  { key: '3', label: '3', test: n => n === 3 },
  { key: '4', label: '4', test: n => n === 4 },
  { key: '5-6', label: '5–6', test: n => n >= 5 && n <= 6 },
  { key: '6+', label: '6+', test: n => n > 6 },
];
export const colourBandOf = row => {
  const n = totalColoursOf(row);
  if (n == null) return null;
  return COLOUR_BANDS.find(b => b.test(n))?.key || null;
};

// ---- Badges ----------------------------------------------------------------
// One component per axis with mode flags, never variants:
//   band    — the card face: full width, centred
//   compact — drops the count, for rails too tight for the whole phrase
// Both always render their text. An unknown/absent value stays silent (returns
// null) so a stale payload paints nothing rather than a wrong badge.

export function ColourBadge({ row, band, compact, className = '' }) {
  const type = colourTypeOf(row);
  if (!type) return null;
  const Icon = COLOUR_ICON[type];
  const total = totalColoursOf(row);
  const text = compact || total == null ? COLOUR_LABEL[type] : `${COLOUR_LABEL[type]} · ${total}`;
  const title = `${colourSummary(row)}${row?.pantone_codes ? ` — ${row.pantone_codes}` : ''}\n${COLOUR_HINT[type]}`;
  return band ? (
    <div title={title}
      className={`mt-1 flex items-center justify-center gap-1.5 rounded-lg border px-2 py-1 text-[11.5px] font-extrabold uppercase tracking-[0.02em] ${COLOUR_TONE[type]} ${className}`}>
      <Icon size={13} className="shrink-0" /> {text}
    </div>
  ) : (
    <span title={title}
      className={`inline-flex items-center gap-1.5 whitespace-nowrap rounded-full border px-2 py-0.5 text-[11px] font-bold ${COLOUR_TONE[type]} ${className}`}>
      <Icon size={12} className="shrink-0" /> {text}
    </span>
  );
}

export function ProcessBadge({ row, band, compact, className = '' }) {
  const p = processOf(row);
  if (!p) return null;
  // Offset is the plant default and says nothing worth a chip in a tight rail.
  if (compact && p === 'Offset') return null;
  const Icon = PROCESS_ICON[p];
  const name = metallicNameOf(row);
  const text = !compact && name && p !== 'Offset' ? `${PROCESS_LABEL[p]} · ${name}` : PROCESS_LABEL[p];
  const title = `${PROCESS_HINT[p]}${row?.metallic_details ? `\n${row.metallic_details}` : ''}`;
  return band ? (
    <div title={title}
      className={`mt-1 flex items-center justify-center gap-1.5 rounded-lg border px-2 py-1 text-[11.5px] font-extrabold uppercase tracking-[0.02em] ${PROCESS_TONE[p]} ${className}`}>
      <Icon size={13} className="shrink-0" /> {text}
    </div>
  ) : (
    <span title={title}
      className={`inline-flex items-center gap-1.5 whitespace-nowrap rounded-full border px-2 py-0.5 text-[11px] font-bold ${PROCESS_TONE[p]} ${className}`}>
      <Icon size={12} className="shrink-0" /> {text}
    </span>
  );
}

// Both axes side by side — the default way a queue row wears its ink.
export function PrintColourChips({ row, compact, className = '' }) {
  const type = colourTypeOf(row);
  const p = processOf(row);
  if (!type && !p) return null;
  return (
    <span className={`inline-flex flex-wrap items-center gap-1 ${className}`}>
      <ColourBadge row={row} compact={compact} />
      <ProcessBadge row={row} compact={compact} />
    </span>
  );
}

// The long detail, for a form panel or an expandable row.
export function ColourDetailLines(row) {
  const out = [];
  const type = colourTypeOf(row);
  if (type) out.push(['Colour Type', type]);
  const total = totalColoursOf(row);
  if (total != null) out.push(['Total Colours', String(total)]);
  const c = cmykCountOf(row); if (c) out.push(['CMYK Colours', String(c)]);
  const pc = pantoneCountOf(row); if (pc) out.push(['Pantone Colours', String(pc)]);
  if (row?.pantone_codes) out.push(['Pantone Codes', row.pantone_codes]);
  const p = processOf(row);
  if (p) out.push(['Printing Process', p]);
  const m = metallicCountOf(row); if (m) out.push(['Metallic Colours', String(m)]);
  if (row?.metallic_details) out.push(['Metallic Details', row.metallic_details]);
  if (row?.print_instructions) out.push(['Printing Instructions', row.print_instructions]);
  return out;
}
```

```js
// Printing colour + process — the server's half of the vocabulary.
//
// Pure functions only, no DB. The client twin is
// client/src/components/PrintColour.jsx and the two MUST agree on derivation;
// the tests here are what stop them drifting.
//
// Validation is deliberately SOFT. Physics is hard, paperwork is soft: a
// missing Pantone code must warn loudly and block nothing, because the plant
// has to be able to plan a job while the customer is still deciding the shade.

export const PRINT_COLOUR_FIELDS = [
  'print_process', 'cmyk_colours', 'pantone_colours', 'pantone_codes',
  'metallic_colours', 'metallic_details', 'print_instructions',
];
export const PRINT_COLOUR_INT_FIELDS = ['cmyk_colours', 'pantone_colours', 'metallic_colours'];
export const PRINT_COLOUR_TEXT_FIELDS = ['print_process', 'pantone_codes', 'metallic_details', 'print_instructions'];

export const COLOUR_TYPES = ['CMYK', 'Pantone', 'CMYK + Pantone'];
export const PRINT_PROCESSES = ['Offset', 'Metallic', 'Offset + Metallic'];

const s = v => String(v ?? '').trim();
const n = v => (v == null || v === '' ? null : Number(v));

export function colourTypeOf(row) {
  const raw = s(row?.colour_type).toLowerCase();
  if (!raw) return null;
  const hasC = raw.includes('cmyk') || raw.includes('process');
  const hasP = raw.includes('pantone') || raw.includes('spot');
  if (hasC && hasP) return 'CMYK + Pantone';
  if (hasP) return 'Pantone';
  if (hasC) return 'CMYK';
  return null;
}

export function processOf(row) {
  const raw = s(row?.print_process).toLowerCase();
  if (raw) {
    const hasM = raw.includes('metallic');
    const hasO = raw.includes('offset');
    if (hasM && hasO) return 'Offset + Metallic';
    if (hasM) return 'Metallic';
    if (hasO) return 'Offset';
  }
  if ((Number(row?.metallic_colours) || 0) > 0 || s(row?.metallic_details)) return 'Offset + Metallic';
  return raw ? null : 'Offset';
}

// The counts a screen should show. Typed values always win; anything NULL is
// inferred from colour_type + colors so a legacy row still reads correctly.
export function derivedCounts(row) {
  const type = colourTypeOf(row);
  const metallicExplicit = n(row?.metallic_colours);
  const metallic = metallicExplicit != null && Number.isFinite(metallicExplicit)
    ? metallicExplicit
    : (s(row?.metallic_details) && processOf(row) !== 'Offset' ? 1 : 0);

  const cmykExplicit = n(row?.cmyk_colours);
  const cmyk = cmykExplicit != null && Number.isFinite(cmykExplicit)
    ? cmykExplicit
    : (!type || type === 'Pantone' ? 0 : 4);

  const totalExplicit = n(row?.colors);
  const pantoneExplicit = n(row?.pantone_colours);
  let pantone;
  if (pantoneExplicit != null && Number.isFinite(pantoneExplicit)) pantone = pantoneExplicit;
  else if (!type || type === 'CMYK') pantone = 0;
  else if (totalExplicit == null) pantone = 0;
  else pantone = Math.max(0, totalExplicit - (type === 'Pantone' ? 0 : 4) - metallic);

  const total = totalExplicit != null && Number.isFinite(totalExplicit)
    ? totalExplicit
    : (cmyk + pantone + metallic) || null;

  return { cmyk, pantone, metallic, total };
}

// Soft warnings. Never a save gate — the caller decides how loudly to say them.
export function printColourWarnings(row) {
  const out = [];
  if (!row || typeof row !== 'object') return out;
  const type = colourTypeOf(row);
  const proc = s(row.print_process) ? processOf(row) : null;
  const counts = derivedCounts(row);

  if ((type === 'Pantone' || type === 'CMYK + Pantone') && !s(row.pantone_codes)) {
    out.push({ code: 'PANTONE_NO_CODES', field: 'pantone_codes',
      message: 'Pantone is selected but no Pantone code or name is entered.' });
  }
  if ((proc === 'Metallic' || proc === 'Offset + Metallic') && !s(row.metallic_details)) {
    out.push({ code: 'METALLIC_NO_DETAIL', field: 'metallic_details',
      message: 'Metallic printing is selected but no metallic colour is entered.' });
  }
  // Only a TYPED total can disagree with typed parts — an inferred one cannot.
  const typedTotal = n(row.colors);
  const anyTypedPart = PRINT_COLOUR_INT_FIELDS.some(f => n(row[f]) != null);
  if (typedTotal != null && anyTypedPart) {
    const sum = (Number(row.cmyk_colours) || 0) + (Number(row.pantone_colours) || 0) + (Number(row.metallic_colours) || 0);
    if (sum !== typedTotal) {
      out.push({ code: 'TOTAL_MISMATCH', field: 'colors',
        message: `Total Colours is ${typedTotal} but CMYK + Pantone + metallic add to ${sum}.` });
    }
  }
  // A metallic ink named against a plain-Offset process. NOTE: a Pantone code is
  // never evidence of metallic — that conflation is exactly what this split fixes.
  if (proc === 'Offset' && (s(row.metallic_details) || (Number(row.metallic_colours) || 0) > 0)) {
    out.push({ code: 'METALLIC_WITHOUT_PROCESS', field: 'print_process',
      message: 'A metallic colour is entered but the Printing Process is set to Offset only.' });
  }
  return out;
}

// The one-line summary the server puts in exports and the printed traveler.
export function colourSummary(row) {
  const type = colourTypeOf(row);
  const { pantone, metallic, total } = derivedCounts(row);
  if (!type && total == null) return '—';
  const bits = [];
  if (type === 'CMYK' || type === 'CMYK + Pantone') bits.push('CMYK');
  if (pantone > 0) bits.push(`${pantone} Pantone`);
  else if (type === 'Pantone') bits.push('Pantone');
  if (metallic > 0) bits.push(`${metallic} Metallic`);
  const head = bits.length ? bits.join(' + ') : (type || '—');
  return total != null ? `${head} — ${total} ${total === 1 ? 'colour' : 'colours'}` : head;
}
```
